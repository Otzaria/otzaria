cmake_minimum_required(VERSION 3.16)
project(otzaria_domless_probe C)
if(NOT QUICKJS_SOURCE)
  message(FATAL_ERROR "Run prepare.py with an external cache directory first")
endif()
set(QJS_BUILD_LIBC OFF CACHE BOOL "No guest filesystem or OS modules" FORCE)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "Standalone experiment executable" FORCE)
add_subdirectory(${QUICKJS_SOURCE} quickjs EXCLUDE_FROM_ALL)
add_executable(otzaria_domless_probe host.c)
target_compile_features(otzaria_domless_probe PRIVATE c_std_11)
target_link_libraries(otzaria_domless_probe PRIVATE qjs)
if(WIN32)
  target_link_libraries(otzaria_domless_probe PRIVATE psapi)
endif()
if(MSVC)
  target_compile_options(otzaria_domless_probe PRIVATE /W4)
else()
  target_compile_options(otzaria_domless_probe PRIVATE -Wall -Wextra -Werror)
endif()
